cmake_minimum_required(VERSION 3.20)
project(uotan_media_va VERSION 1.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The VA-API bridge is Linux-only")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec>=59 libavutil>=57)
add_library(uotan_media_va SHARED uotan_va.c)
target_link_libraries(uotan_media_va PRIVATE PkgConfig::AV)
target_compile_features(uotan_media_va PRIVATE c_std_11)
target_compile_options(uotan_media_va PRIVATE -Wall -Wextra -Werror)
set_target_properties(uotan_media_va PROPERTIES C_VISIBILITY_PRESET hidden)
include(GNUInstallDirs)
install(TARGETS uotan_media_va LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
include(CTest)
if(BUILD_TESTING)
    add_executable(va_bridge_tests tests/bridge_tests.c)
    target_link_libraries(va_bridge_tests PRIVATE PkgConfig::AV)
    target_compile_options(va_bridge_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME va_bridge_contract COMMAND va_bridge_tests
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../UotanStudio.Tests/TestInfrastructure/Media/black-64x64.h264")
endif()
